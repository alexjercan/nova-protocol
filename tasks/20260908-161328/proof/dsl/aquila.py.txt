def dialogue_lines(source):
    """Read whole dialogue paragraphs, including source-wrapped continuation lines."""
    return [(speaker,' '.join(line.split())) for speaker,line in re.findall(
        r'^\*\*([^*\n]+):\*\* ([\s\S]+?)(?=\n\n|\Z)',source,re.M
    ) if speaker!='Purpose']


def script_panels():
    """Read dialogue from the script; page geometry owns wrapping, not a second script."""
    source = (OUTPUT/'SCRIPT.md').read_text()
    panels = {}
    for match in re.finditer(r'^### Panel ([567][abc])\n(.*?)(?=^###? |\Z)',source,re.M|re.S):
        key,body = match.groups()
        lines = dialogue_lines(body)
        action = ' '.join(re.split(r'^\*\*[^\n]+:\*\* ',body,maxsplit=1,flags=re.M)[0].split()).replace('**','')
        panels[key] = (action,lines)
    assert len(panels)==9
    return panels


class Panel(opening.Panel):
    """Retain the opening frame and text style with explicitly sided balloon tails."""

    def __init__(self,key,x,y,w,h,art,script):
        action,self.spoken = script[key]
        super().__init__(key,x,y,w,h,action,art)
        self.used = []

    def say_line(self,index,x,y,w,wrap,tip,side='bottom'):
        """Lay out one script line and remember its use for exact transcript checks."""
        speaker,line = self.spoken[index]
        lines = textwrap.wrap(line,wrap,break_long_words=False,break_on_hyphens=False)
        self.balloons.append({'x':self.x+x,'y':self.y+y,'width':w,'speaker':speaker,'text':'\n'.join(lines),'tip':[self.x+tip[0],self.y+tip[1]],'side':side})
        self.lettering += labelled_speech(x,y,w,speaker,lines,tip,side)
        self.lines.append((speaker,line))
        self.used.append(index)
        return self

    def speak(self,index,x,y,w,tip,side='bottom'):
        """Let the reader measure and wrap a whole script line in an authored region."""
        speaker,line = self.spoken[index]
        self.balloons.append({'x':self.x+x,'y':self.y+y,'width':w,'speaker':speaker,'text':line,'tip':[self.x+tip[0],self.y+tip[1]],'side':side})
        self.lines.append((speaker,line))
        self.used.append(index)
        return self

    def render(self):
        """Reject omitted, repeated, or reordered dialogue before saving a page."""
        assert self.used==list(range(len(self.spoken))),self.key
        return super().render()


def window(key,x,y,w,h,view):
    """Keep space behind a framed pressure window, never an open cargo doorway."""
    inside = rect(x,y,w,h,'url(#space)')+view
    clip = tag('defs',tag('clipPath',rect(x,y,w,h,'white',radius=30),id=key))
    art = clip+group(inside,clip_path=f'url(#{key})')
    art += rect(x,y,w,h,'none',INK,14,30)
    art += rect(x,y,w,h,'none',FREIGHT['rail'],8,30)
    art += path(f'M{x+18} {y+h-22}V{y+43}Q{x+18} {y+18} {x+45} {y+18}',stroke=FREIGHT['edge'],width=2)
    return group(art,data_prop='pressure-window')


def held_person(name,x,y,scale,reach=False):
    """Anchor a frontal person at a drawn rail without rotating their likeness."""
    body = gantry_portrait(name) if name=='nadia' else jonah_listener() if name=='jonah' else work_portrait(name)
    art = group(body,f'translate({x} {y}) scale({scale})')+handhold(x,y,scale)
    arm = reaching_arm(name) if reach else gripping_arm(name)
    return art+group(arm,f'translate({x} {y}) scale({scale})')


def clear_signal(x,y):
